Финальный проект: TODO-приложение

~45 минут Продвинутый уровень Теория и Практика

Это финальный урок курса. Мы соберём приложение целиком: от структуры файлов до сохранения задач между перезагрузками. Здесь встретится всё, что мы прошли: DOM, события, массивы, методы, модули, ошибки и хранилище.

В этом уроке мы:

  • Спроектируем структуру проекта и состояние.
  • Напишем рендер списка задач.
  • Добавим добавление, переключение и удаление.
  • Сделаем фильтры и сохранение в браузере.
Аналогия

Кухня: состояние - продукты, render - готовка, события - заказы официанта. Повар не бегает в зал, а работает с тем, что принесли.

Что получится

Приложение «Список задач»: поле ввода, список, три фильтра и счётчик. Задачи сохраняются в localStorage, поэтому после перезагрузки ничего не пропадает.

Структура проекта

todo/├── index.html├── css/│   └── style.css└── js/    ├── main.js       точка входа    ├── storage.js    чтение и запись в localStorage    └── render.js     отрисовка списка

Три файла - не «оверинжиниринг», а граница ответственности. Хранилище не знает про DOM, отрисовка не знает, откуда взялись данные. Менять и тестировать такое приложение заметно проще.

Шаг 2. Хранилище

// js/storage.jsconst KEY = "coderush-todos";export function loadTodos() {    try {        const raw = localStorage.getItem(KEY);        if (!raw) return [];        const parsed = JSON.parse(raw);        return Array.isArray(parsed) ? parsed : [];    } catch (error) {        console.warn("Не удалось прочитать задачи:", error.message);        return [];    }}export function saveTodos(todos) {    try {        localStorage.setItem(KEY, JSON.stringify(todos));    } catch (error) {        console.warn("Не удалось сохранить задачи:", error.message);    }}

Два уровня защиты: от повреждённого JSON и от переполненного хранилища. Приложение не должно падать из-за проблем с localStorage.

Шаг 4. Рендер

JavaScript 
// js/render.js
export function renderTodos(elements, todos, filter) {
    const visible = todos.filter((todo) => {
        if (filter === "active") return !todo.done;
        if (filter === "done") return todo.done;
        return true;
    });

    elements.list.innerHTML = "";

    visible.forEach((todo) => {
        const item = document.createElement("li");
        item.className = "todo-item";
        item.classList.toggle("is-done", todo.done);
        item.dataset.id = todo.id;

        const label = document.createElement("label");
        label.className = "todo-label";

        const checkbox = document.createElement("input");
        checkbox.type = "checkbox";
        checkbox.checked = todo.done;
        checkbox.className = "todo-toggle";

        const text = document.createElement("span");
        text.className = "todo-text";
        text.textContent = todo.text;

        label.append(checkbox, text);

        const remove = document.createElement("button");
        remove.type = "button";
        remove.className = "todo-remove";
        remove.textContent = "×";
        remove.setAttribute("aria-label", "Удалить задачу");

        item.append(label, remove);
        elements.list.append(item);
    });

    elements.empty.hidden = visible.length > 0;
    elements.summary.textContent = summary(todos);
}

function summary(todos) {
    const done = todos.filter((todo) => todo.done).length;
    return `Выполнено ${done} из ${todos.length}`;
}

Обратите внимание: текст задачи вставляется через textContent, а не innerHTML. Пользователь может ввести <script> - и это останется текстом, а не кодом.

Шаг 5. Действия

JavaScript 
// js/main.js
function addTodo(text) {
    const todo = {
        id: crypto.randomUUID(),
        text: text.trim(),
        done: false
    };

    state.todos = [...state.todos, todo];
    update();
}

function toggleTodo(id) {
    state.todos = state.todos.map((todo) =>
        todo.id === id ? { ...todo, done: !todo.done } : todo
    );
    update();
}

function removeTodo(id) {
    state.todos = state.todos.filter((todo) => todo.id !== id);
    update();
}

function update() {
    saveTodos(state.todos);
    render();
}

Все три функции меняют состояние неизменяемо: создают новые массивы и объекты. Ни одна не трогает DOM напрямую - этим занимается render.

crypto.randomUUID

Современный способ получить уникальный идентификатор. Он встроен в браузер и не требует библиотек. В старых окружениях используют счётчик или Date.now() - но с ростом приложения коллизии становятся реальной проблемой.

Шаг 6. Связываем всё

JavaScript 
// js/main.js
import { loadTodos, saveTodos } from "./storage.js";
import { renderTodos } from "./render.js";

const state = {
    todos: loadTodos(),
    filter: "all"
};

const elements = {
    form: document.querySelector(".todo-form"),
    input: document.querySelector(".todo-input"),
    list: document.querySelector(".todo-list"),
    empty: document.querySelector(".todo-empty"),
    summary: document.querySelector(".todo-summary"),
    filters: document.querySelector(".todo-filters")
};

function render() {
    renderTodos(elements, state.todos, state.filter);
}

function update() {
    saveTodos(state.todos);
    render();
}

function addTodo(text) {
    const todo = {
        id: crypto.randomUUID(),
        text: text.trim(),
        done: false
    };

    state.todos = [...state.todos, todo];
    update();
}

function toggleTodo(id) {
    state.todos = state.todos.map((todo) =>
        todo.id === id ? { ...todo, done: !todo.done } : todo
    );
    update();
}

function removeTodo(id) {
    state.todos = state.todos.filter((todo) => todo.id !== id);
    update();
}

elements.form.addEventListener("submit", (event) => {
    event.preventDefault();

    const text = elements.input.value.trim();
    if (!text) return;

    addTodo(text);
    elements.form.reset();
    elements.input.focus();
});

elements.list.addEventListener("click", (event) => {
    const item = event.target.closest(".todo-item");
    if (!item) return;

    const id = item.dataset.id;

    if (event.target.classList.contains("todo-toggle")) {
        toggleTodo(id);
    }

    if (event.target.classList.contains("todo-remove")) {
        removeTodo(id);
    }
});

elements.filters.addEventListener("click", (event) => {
    const button = event.target.closest("[data-filter]");
    if (!button) return;

    state.filter = button.dataset.filter;

    elements.filters.querySelectorAll("[data-filter]").forEach((btn) => {
        btn.classList.toggle("is-active", btn === button);
    });

    render();
});

render();

Разберём ключевые решения:

  • Один обработчик на список. Делегирование: задачи добавляются и удаляются, но обработчик навешен один раз и работает для всех.
  • closest защищает от клика по вложенному элементу.
  • dataset.id связывает DOM-элемент с записью в состоянии.
  • event.preventDefault() не даёт форме перезагрузить страницу.
  • Финальный render() отрисовывает сохранённые задачи при загрузке.

Как проверить

Из-за ES-модулей открыть index.html двойным щелчком не получится: браузер заблокирует импорты на file://. Запустите локальный сервер из папки проекта:

python -m http.server 8000

И откройте http://localhost:8000. Проверьте сценарий:

  1. Добавьте три задачи.
  2. Отметьте одну выполненной.
  3. Переключите фильтры - список должен меняться.
  4. Перезагрузите страницу - задачи и статусы на месте.
  5. Удалите задачу - счётчик обновился.

Что можно улучшить

Проект рабочий, но это только начало. Вот естественные следующие шаги:

  • Редактирование. Двойной клик по тексту - и он превращается в поле ввода.
  • Сортировка. Новые задачи сверху или выполненные в конце.
  • Массовые действия. «Очистить выполненные» одной кнопкой.
  • Клавиатура. Escape отменяет редактирование, Enter сохраняет.
  • Доступность. aria-label, работа с экранным диктором, видимый фокус.
  • Синхронизация. Отправка задач на сервер через fetch.

Куда двигаться дальше

Вы прошли путь от console.log до полноценного приложения. Что изучать следующим:

  • TypeScript. Типы поверх JavaScript. Ловит половину ошибок ещё до запуска кода и обязателен в большинстве команд.
  • React, Vue или Svelte. Фреймворки, которые автоматизируют то, что вы делали руками: рендер из состояния и обновление DOM.
  • Node.js. Тот же язык на сервере: API, базы данных, авторизация.
  • Тестирование. Vitest или Jest - чтобы изменения не ломали уже работающее.
  • Инструменты. Git, npm, сборщики (Vite), линтеры (ESLint) и форматтеры (Prettier).
Главный совет

Не пытайтесь выучить всё сразу. Возьмите одно приложение и доведите его до конца: с обработкой ошибок, пустыми состояниями и понятным интерфейсом. Один законченный проект учит больше, чем десять начатых курсов.

Шпаргалка

Всё приложение держится на пяти понятиях:

  • state единственный источник данных
  • render() рисует интерфейс из состояния
  • update() сохраняет и перерисовывает
  • localStorage переживает перезагрузку
  • closest делегирование событий

Если что-то не работает

  • «После перезагрузки пусто». Забыли вызвать render() при старте или ' 'не сохранили состояние в localStorage.
  • «Задача удаляется, но возвращается». Изменение applied к копии массива, а не ' 'к state.todos. Проверьте, что результат filter присвоен обратно.
  • «Клик по кнопке удаляет не ту задачу». dataset.id читается ' 'у соседнего элемента: ищите closest(".todo-item") от event.target.
  • «Счётчик не совпадает со списком». Считаете по DOM, а не по состоянию. ' 'Любые цифры должны выводиться из state.todos.
  • «Задачи пропали после смены браузера». localStorage привязан ' 'к конкретному браузеру и устройству. Для синхронизации нужен сервер.
  • После перезагрузки всё пропало. Состояние жило только в памяти. Любое приложение, которое обещает «сохранить», должно писать данные в хранилище или на сервер - иначе закрытая вкладка стирает работу.

Практика

Задача 1. Очистить выполненные

Добавьте кнопку «Очистить выполненные», которая удаляет все задачи с done: true. Кнопка должна быть неактивна, если таких задач нет.

Подсказка

Отфильтруйте состояние: state.todos.filter(t => !t.done). Для неактивного состояния используйте свойство disabled у кнопки, обновляя его в render().

Решение учителя 
function clearCompleted() {    state.todos = state.todos.filter((todo) => !todo.done);    update();}// в render():const hasDone = state.todos.some((todo) => todo.done);elements.clearBtn.disabled = !hasDone;
Задача 2. Редактирование по двойному клику

При двойном клике по тексту задачи замените его на поле ввода. Enter сохраняет новое значение, Escape отменяет изменения.

Подсказка

Слушайте dblclick на списке через делегирование. Замените span на input, сохраните в keydown и не забудьте вызвать update(), чтобы состояние и хранилище обновились.

Решение учителя
JavaScript 
function editTodo(id, text) {
    state.todos = state.todos.map((todo) =>
        todo.id === id ? { ...todo, text } : todo
    );
    update();
}

elements.list.addEventListener("dblclick", (event) => {
    const textEl = event.target.closest(".todo-text");
    if (!textEl) return;

    const item = textEl.closest(".todo-item");
    const input = document.createElement("input");
    input.value = textEl.textContent;

    textEl.replaceWith(input);
    input.focus();

    input.addEventListener("keydown", (event) => {
        if (event.key === "Enter") {
            editTodo(item.dataset.id, input.value.trim() || textEl.textContent);
        }
        if (event.key === "Enter" || event.key === "Escape") {
            render();
        }
    });
});
